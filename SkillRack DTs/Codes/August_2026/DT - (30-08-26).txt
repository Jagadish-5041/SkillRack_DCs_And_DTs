Problem Statement :-

Even/Odd Based On Integer Digits in DT (Daily Test) in SkillRack in Python - Programming

Code :-

def is_odd(N):
    
    return ((N & 1) == 1)

def is_even(N):
    
    return ((N & 1) == 0)

N = int(input().strip())

odd_digits_count = 0

even_digits_count = 0

while(N > 0):
    
    digit = (N % 10)
    
    odd_digits_count = (odd_digits_count + 1) if(is_odd(digit)) else (odd_digits_count)
    
    even_digits_count = (even_digits_count + 1) if(is_even(digit)) else (even_digits_count)
    
    N = (N // 10)

print(("odd") if(odd_digits_count > even_digits_count) else (("even") if(even_digits_count > odd_digits_count) else (-1)))
