def get_parity_bit(N):
    
    return (N & 1)

def print_target_parity_bit_values(target_parity_bit , values_list):
    
    print(*[value for value in values_list if(get_parity_bit(value) == target_parity_bit)] , sep = " " , end = " ")

N = int(input().strip())
