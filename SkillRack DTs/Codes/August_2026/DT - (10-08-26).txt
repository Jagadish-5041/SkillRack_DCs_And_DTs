Problem Statement :-

Largest Median (Error Identification) in DT (Daily Test) in SkillRack in C - Programming

Code :-

#include <stdio.h>
#include <stdlib.h>

int main(){
    int N;
    int value;
    int median;
    int totalValue;
    scanf("%d %d",&N,&value);
    median = value;
    totalValue = value;
    for(int ctr=2;ctr<=N;ctr++){ // Error - 1 { (ctr+1) is changed to (ctr++) }
        scanf("%d ",&value);
        totalValue += value; // Error - 2 { (=+) is changed to (+=) }
        if(median >= (totalValue / ctr)){ // Error - 3 { (%) is changed to (/) }
            median = (totalValue / ctr);
        }
    }
    printf("%d ",median);
    return 0;
}
