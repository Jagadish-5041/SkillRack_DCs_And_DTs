def print_lowercase_alphabets_within_range(starting_alphabet , ending_alphabet):
    
    for lowercase_alphabet_ASCII_value in range((ord(starting_alphabet) + 1) , ((ord(ending_alphabet) - 1) + 1)):
        
        print(chr(lowercase_alphabet_ASCII_value) , end = "")
