def is_boundary(row_index , column_index , R , C):
    
    return ((row_index == 0) or (row_index == (R - 1)) or (column_index == 0) or (column_index == (C - 1)))

R , C = map(int , input().strip().split())

matrix = [list(map(int , input().strip().split())) for i in range(0 , R)]

[print(*[(matrix[i][j]) if(is_boundary(i , j , R , C)) else ('*') for j in range(0 , C)]) for i in range(0 , R)]
