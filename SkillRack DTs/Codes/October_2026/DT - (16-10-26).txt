def get_smallest_digit(N):
    
    smallest_digit = 10
    
    while(N > 0):
        
        current_digit = (N % 10)
        
        smallest_digit = min(smallest_digit , current_digit)
        
        N = (N // 10)
    
    return smallest_digit
