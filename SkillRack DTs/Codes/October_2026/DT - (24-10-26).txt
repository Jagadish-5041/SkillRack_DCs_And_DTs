S = input().strip()

left_pointer = 0

right_pointer = (len(S) - 1)

while(left_pointer <= right_pointer):
    
    (print(S[left_pointer] , end = "")) if(left_pointer == right_pointer) else (print(S[left_pointer] , S[right_pointer] , sep = "" , end = ""))
    
    if(S[left_pointer] == S[right_pointer]):
        break
