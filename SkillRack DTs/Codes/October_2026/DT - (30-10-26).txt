Problem Statement :-

Print From Last Odd Integer in DT (Daily Test) in SkillRack in Python - Programming

Code :-

def is_odd(N):
    
    return ((N & 1) == 1)

N = int(input().strip())

values_list = list(map(int , input().strip().split()))

print_flag = False

for index in range((N - 1) , -1 , -1):
    
    print_flag = (True) if(is_odd(values_list[index])) else (print_flag)
    
    (print(values_list[index] , end = " ")) if(print_flag) else ("")
