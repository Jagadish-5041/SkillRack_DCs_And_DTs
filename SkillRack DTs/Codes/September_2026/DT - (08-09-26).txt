def get_interlaced_data_list(A_N , A , B_N , B):
    
    C = []
    
    [(C.append((X , 1)) , C.append((Y , 2))) for X , Y in zip(A , B)]
    
    ([C.append((X , 1)) for X in A[B_N : ]]) if(A_N > B_N) else ([C.append((Y , 2)) for Y in B[A_N : ]])
    
    return C
