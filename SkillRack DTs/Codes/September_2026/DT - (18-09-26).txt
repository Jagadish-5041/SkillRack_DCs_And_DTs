def inverted_u_pattern_printing(top,vertical_left_string,vertical_right_string):
    print((*top),sep=" ")
    for i in range(1,len(vertical_left_string)):
        for j in range(0,len(top)):
            print(vertical_left_string[i],end=" ") if(j==0) else (print(vertical_right_string[i],end=" ") if(j==(len(top) - 1)) else print("*",end=" "))
        print()
