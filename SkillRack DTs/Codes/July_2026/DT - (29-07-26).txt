def is_within_bounds(index , N):
    
    return (0 <= index < N)

def diagonal_traversal_function(fill_or_fetch_flag , current_i , current_j , offset_i , offset_j , R , C):
    
    global diagonally_traversed_elements_list
    
    global diagonally_traversed_elements_list_index
    
    global matrix
    
    while((is_within_bounds(current_i , R)) and (is_within_bounds(current_j , C))):
        
        if(fill_or_fetch_flag == 1):
            diagonally_traversed_elements_list.append(matrix[current_i][current_j])
        else:
            matrix[current_i][current_j] = diagonally_traversed_elements_list[diagonally_traversed_elements_list_index]
            diagonally_traversed_elements_list_index = (diagonally_traversed_elements_list_index + 1)
        
        current_i = (current_i + offset_i)
        
        current_j = (current_j + offset_j)
