def get_3_digit_number(digit_1 , digit_2 , digit_3):
    
    return ((((digit_1 * 10) + digit_2) * 10) + digit_3)

def is_divisible_by_8(N):
    
    last_3_digits_number = (N % 1000)
    
    if((last_3_digits_number % 8) == 0):
        return True
    
    last_3_digits_list = [(last_3_digits_number % 10) , ((last_3_digits_number % 100) // 10) , ((last_3_digits_number % 1000) // 100)]
    
    for i in range(0 , 3):
        
        for j in range(0 , 3):
            
            for k in range(0 , 3):
                
                if((i != j) and (j != k) and (i != k) and ((get_3_digit_number(last_3_digits_list[i] , last_3_digits_list[j] , last_3_digits_list[k]) % 8) == 0)):
                    
                    return True
    
    return False
