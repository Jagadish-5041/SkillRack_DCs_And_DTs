def is_consonant(character):
    
    return ((character.isalpha()) and (character not in "aeiouAEIOU"))

def get_consonants_removed_string(string):
    
    return ("".join(character for character in string if(not is_consonant(character))))

def is_palindrome(N , string):
    
    for index in range(0 , ((N // 2) + (N % 2))):
        
        if(string[index] != string[(N - index - 1)]):
            
            return False
    
    return True
