Problem Statement :-

N Terms - Fraction Series in DC (Daily Challenge) in SkillRack in C - Programming

Code :-

#include<stdio.h>
#include<stdlib.h>

int main()
{
    int a;
    
    scanf("%d",&a);
    
    for(int i=1;i<=a;i++){
        if(i%4==0){
            printf("%d ",i/4);
        }
        else if(i%2==0){
            printf("%d/2 ",i/2);
        }
        else{
            printf("%d/4 ",i);
        }
    }

}
