First N Terms in DC (Daily Challenge) in SkillRack in Python - Programming

def get_sum_of_N_natural_numbers(N):
    
    return ((N * (N + 1)) // 2)

N = int(input().strip())

running_range_value = 1

series_list = []

for number in range(1 , (N + 1)):
    
    starting_range_value = running_range_value
    
    ending_range_value = (running_range_value + number - 1)
    
    running_range_value = (running_range_value + number)
    
    series_list.append((get_sum_of_N_natural_numbers(ending_range_value) - get_sum_of_N_natural_numbers((starting_range_value - 1))))

print(*series_list)
