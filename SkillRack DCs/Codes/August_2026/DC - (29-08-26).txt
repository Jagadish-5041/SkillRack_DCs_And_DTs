#include <stdio.h>
#include <stdlib.h>

int count_divisors_of_N_which_has_X_as_a_factor(int N , int X){
    
    int count = 0;
    
    for(int number = 1 ; (number <= N) ; number++){
        
        if(((N % number) == 0) && ((number % X) == 0)){
            
            count++;
        }
    }
    
    return count;
}
