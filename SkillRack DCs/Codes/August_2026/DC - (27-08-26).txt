def is_within_bounds(index , N):
    
    return (0 <= index < N)

    global diagonal_elements_list
    
    global diagonal_elements_list_index
    
    while((is_within_bounds(current_i , R)) and (is_within_bounds(current_j , C))):
        
        (diagonal_elements_list.append(matrix[current_i][current_j])) if(get_or_fill_flag_value == 1) else (matrix[current_i].__setitem__(current_j , diagonal_elements_list[diagonal_elements_list_index]))
        
        diagonal_elements_list_index = (diagonal_elements_list_index + 1) if(get_or_fill_flag_value == 2) else (diagonal_elements_list_index)
        
        current_i = (current_i + offset_i)
        
        current_j = (current_j + offset_j)
