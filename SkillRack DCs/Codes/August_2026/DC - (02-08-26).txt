Code :-

#include <stdio.h>
#include <stdlib.h>
#define jagadish_DC main

int jagadish_DC(){
    
    int N;
    
    int maximum_value;
    
    scanf("%d %d\n" , &N , &maximum_value);
    
    for(int current_value = -999 , counter_value = 2 ; (counter_value <= N) ; counter_value++){
        
        scanf("%d " , &current_value);
        
        maximum_value = (maximum_value < current_value) ? (current_value) : (maximum_value);
    }
    
    printf("%d" , maximum_value);
    
    return 0;
}
