def is_border(i , j , R , C):
    
    return ((i == 0) or (i == (R - 1)) or (j == 0) or (j == (C - 1)))

def is_primary_diagonal(i , j , R , C):
    
    return (i == j)

def is_secondary_diagonal(i , j , R , C):
    
    return ((i + j) == (C - 1))

def get_cell_value(i , j , R , C):
    
    return (('*') if((is_border(i , j , R , C)) or (is_primary_diagonal(i , j , R , C)) or (is_secondary_diagonal(i , j , R , C))) else ('-'))
