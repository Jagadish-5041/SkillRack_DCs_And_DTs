Problem Statement :-

Unique Integers (Error Identification) in DC (Daily Challenge) in SkillRack in C - Programming

Code :-

// Program given in Query :-

#include<stdio.h>
#include<stdlib.h>

int main(){
    int N;
    scanf("%d", &N);
    int value, numberCount[1001] = 0;
    for(int ctr = 1; ctr <= N; ctr+1){
        scanf("%d", value);
        if(numberCount[value]++ != 1){
            printf("NO");
            break;
        }
    }
    printf("YES");
    return 0;
}

// After Error Rectification

#include<stdio.h>
#include<stdlib.h>

int main(){
    int N;
    scanf("%d", &N);
    int value, numberCount[1001] = {0}; // Error - 1 { ( 0 ) is changed to ([0]) }
    for(int ctr = 1; ctr <= N; ctr++){ // Error - 2 { (ctr+1) is changed to (ctr++) }
        scanf("%d",&value); // Error - 3 { (value) is changed to (&value) }
        if((++numberCount[value]) > 1){ // Error - 4 { (!=) is changed to (>) } and Error - 5 { (numberCount[value]++) is changed to (++numberCount[value]) }
            printf("NO");
            return 0; // Error - 6 { (break) is changed to (return 0) }
        }
    }
    printf("YES");
    return 0;
}
