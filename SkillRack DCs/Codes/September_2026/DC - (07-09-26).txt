def check_divisibility_of_each_other(A , B):
    
    return (((A % B) == 0) or ((B % A) == 0))

N = int(input().strip())

values_list = list(map(int , input().strip().split()))

count = sum(1 for i in range(0 , (N - 1)) for j in range((i + 1) , N) if(check_divisibility_of_each_other(values_list[i] , values_list[j])))

print(count)
