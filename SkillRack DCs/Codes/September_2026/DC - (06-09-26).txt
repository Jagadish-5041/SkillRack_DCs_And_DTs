Code :-

B , G = map(int , input().strip().split())

dancing_pairs_list = []

[dancing_pairs_list.append((1 , girl)) for girl in range(1 , (G + 1))]

[dancing_pairs_list.append((boy , G)) for boy in range(2 , (B + 1))]

N = len(dancing_pairs_list)

print(N)
