def is_vowel(character):
    
    return ((character.isalpha()) and (character in "aeiouAEIOU"))

def get_offset_value(character):
    
    return ((ord('a')) if(character.islower()) else (ord('A')))

def get_X_th_alphabet_from_current_alphabet(current_alphabet , X):
    
    offset_value = get_offset_value(current_alphabet)
    
    X_th_alphabet = chr((((ord(current_alphabet) - offset_value + X) % 26) + offset_value))
    
    return X_th_alphabet

def get_replaceable_character(character):
    
    if(is_vowel(character)):
        return character
    elif(character == 'z'):
        return 'b'
    elif(is_vowel(get_X_th_alphabet_from_current_alphabet(character , 1))):
        return get_X_th_alphabet_from_current_alphabet(character , 2)
    else:
        return get_X_th_alphabet_from_current_alphabet(character , 1)

print(*[get_replaceable_character(character) for character in input().strip()] , sep = "")
