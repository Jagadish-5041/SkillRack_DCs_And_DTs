def get_sum_of_digits(N):
    
    sum_of_digits = 0
    
    while(N > 0):
        
        digit = (N % 10)
        
        sum_of_digits = (sum_of_digits + digit)
        
        N = (N // 10)
    
    return sum_of_digits
