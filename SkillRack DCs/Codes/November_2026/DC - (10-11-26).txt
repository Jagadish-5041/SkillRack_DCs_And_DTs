import math

def get_length_of_N(N):
    
    return ((0) if(N == 0) else (int(math.log10(abs(N))) + 1))

X , Y = map(int , input().strip().split())

S = (X + Y)

print((S) if(get_length_of_N(S) == X) else (X))
